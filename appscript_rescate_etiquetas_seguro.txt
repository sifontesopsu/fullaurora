function doPost(e) {
  let lock;
  try {
    lock = LockService.getScriptLock();
    // No dejamos a Streamlit esperando 15s si otro POST ya está escribiendo.
    // La app marcará el evento como pendiente y lo reintentará después.
    if (!lock.tryLock(1000)) {
      return json_({ ok: false, transient: true, retry: true, code: "LOCK_BUSY", error: "Apps Script ocupado: otro proceso mantiene el lock" });
    }

    const data = JSON.parse((e && e.postData && e.postData.contents) || "{}");
    const ss = SpreadsheetApp.openById(EVENTS_SPREADSHEET_ID);
    const eventos = getOrCreateSheet_(ss, "eventos", EVENT_HEADERS);
    ensureHeaders_(eventos, EVENT_HEADERS);

    const eventKey = buildEventKey_(data);
    if (eventAlreadyExists_(ss, eventos, eventKey)) {
      return json_({
        ok: true,
        duplicate: true,
        event_key: eventKey,
        event_uid: data.event_uid || "",
        message: "Evento duplicado ignorado por llave estable"
      });
    }

    appendObjectRow_(eventos, {
      received_at: new Date(),
      event_uid: data.event_uid || "",
      event_source: data.event_source || "",
      event_type: data.event_type || "",
      queue_id: data.queue_id || "",
      queued_at: data.queued_at || "",
      lote_id: data.lote_id || "",
      lote_nombre: data.lote_nombre || "",
      backup_lote_key: data.backup_lote_key || "",
      lote_created_at: data.lote_created_at || "",
      archivo: data.archivo || "",
      hoja: data.hoja || "",
      item_id: data.item_id || "",
      codigo_ml: data.codigo_ml || "",
      codigo_universal: data.codigo_universal || "",
      sku: data.sku || "",
      descripcion: data.descripcion || "",
      descripcion_kame: data.descripcion_kame || data.descripcion || "",
      descripcion_ml: data.descripcion_ml || "",
      descripcion_fuente: data.descripcion_fuente || "",
      familia_kame: data.familia_kame || "",
      maestro_match_status: data.maestro_match_status || "",
      cantidad: data.cantidad || "",
      unidades: data.unidades || "",
      modo: data.modo || "",
      tipo: data.tipo || "",
      comentario: data.comentario || data.detail || "",
      event_key: eventKey,
      raw_json: JSON.stringify(data)
    });

    rememberEventKey_(ss, data, eventKey);

    const warnings = [];
    try {
      writeStructured_(ss, data, eventKey);
    } catch (structuredErr) {
      warnings.push(String(structuredErr));
      logError_("writeStructured_", structuredErr, e);
    }

    return json_({ ok: true, event_key: eventKey, event_uid: data.event_uid || "", warnings: warnings });

  } catch (err) {
    logError_("doPost", err, e);
    return json_({ ok: false, error: String(err) });
  } finally {
    if (lock) {
      try { lock.releaseLock(); } catch (err) {}
    }
  }
}

function doGet(e) {
  try {
    const action = (e && e.parameter && e.parameter.action) || "";
    const ss = SpreadsheetApp.openById(EVENTS_SPREADSHEET_ID);

    // Catálogo liviano: no lee scans, snapshots, picking ni auditoría completos.
    if (action === "lotes") {
      return json_({ ok: true, lotes: buildLoteCatalog_(ss) });
    }

    // Rescate paginado desde la hoja madre. Las hojas estructuradas son auditoría y
    // fallback, pero no se vuelven a duplicar dentro de la descarga normal.
    if (action === "events") {
      return json_(readMainEventsPage_(ss, (e && e.parameter) || {}));
    }

    // Conciliación de timeouts: Python solo necesita UID/llave, no todo el historial.
    if (action === "event_keys") {
      const limit = clampInt_((e && e.parameter && e.parameter.limit) || 5000, 100, 20000);
      return json_({ ok: true, event_keys: readRecentEventKeys_(ss, limit) });
    }

    if (action === "reindex_event_keys") {
      const eventos = getOrCreateSheet_(ss, "eventos", EVENT_HEADERS);
      const keySheet = getOrCreateSheet_(ss, "event_keys", EVENT_KEY_HEADERS);
      const added = ensureEventKeyIndex_(eventos, keySheet, true);
      return json_({ ok: true, action: action, added: added });
    }

    return json_({ ok: true, message: "Webhook activo" });
  } catch (err) {
    logError_("doGet", err, e);
    return json_({ ok: false, error: String(err) });
  }
}


function clampInt_(value, minValue, maxValue) {
  const n = parseInt(value, 10);
  if (!isFinite(n)) return minValue;
  return Math.max(minValue, Math.min(maxValue, n));
}

function normalizedText_(value) {
  return String(value === null || value === undefined ? "" : value)
    .replace(/\s+/g, " ")
    .trim()
    .toUpperCase();
}

function timestampText_(value) {
  if (!value) return "";
  if (Object.prototype.toString.call(value) === "[object Date]" && !isNaN(value.getTime())) {
    return value.toISOString();
  }
  return String(value).trim();
}

function eventTimestamp_(ev) {
  return timestampText_(ev.created_at || ev.queued_at || ev.received_at || "");
}

function legacyRescueKey_(ev) {
  return "LEGACY:" + [
    ev.lote_id || "",
    normalizedText_(ev.lote_nombre || ""),
    normalizedText_(ev.archivo || ""),
    normalizedText_(ev.hoja || ""),
    timestampText_(ev.lote_created_at || ev.created_at || ev.queued_at || "")
  ].join("|");
}

function catalogMatch_(records, ev) {
  const explicit = String(ev.backup_lote_key || "").trim();
  if (explicit) {
    for (let i = 0; i < records.length; i++) {
      if (String(records[i].backup_lote_key || "").trim() === explicit) return records[i];
    }
  }
  const lid = String(ev.lote_id || "").trim();
  const name = normalizedText_(ev.lote_nombre || "");
  let matches = records.filter(function(r) {
    return String(r.lote_id || "").trim() === lid && (!name || normalizedText_(r.lote_nombre || "") === name);
  });
  if (!matches.length) return null;
  matches.sort(function(a, b) { return timestampText_(a.created_at).localeCompare(timestampText_(b.created_at)); });
  const ts = eventTimestamp_(ev);
  let chosen = matches[0];
  matches.forEach(function(r) {
    if (!ts || timestampText_(r.created_at) <= ts) chosen = r;
  });
  return chosen;
}

function addCatalogEvent_(catalogMap, ev) {
  const loteId = String(ev.lote_id || "").trim();
  if (!loteId) return;
  const backupKey = String(ev.backup_lote_key || "").trim();
  const rescueKey = backupKey || legacyRescueKey_(ev);
  const createdAt = timestampText_(ev.lote_created_at || ev.created_at || ev.queued_at || ev.received_at || "");
  const rec = catalogMap[rescueKey] || {
    rescue_key: rescueKey,
    backup_lote_key: backupKey,
    lote_id: ev.lote_id || "",
    lote_nombre: ev.lote_nombre || ("Lote " + loteId),
    archivo: ev.archivo || "",
    hoja: ev.hoja || "",
    lote_created_at: createdAt,
    created_at: createdAt,
    ultimo_evento: createdAt,
    total_lineas: ev.total_lineas || ev.productos_total || "",
    total_unidades: ev.total_unidades || ev.unidades_total || "",
    status: ev.status || ev.estado || "ACTIVO"
  };
  if (ev.lote_nombre) rec.lote_nombre = ev.lote_nombre;
  if (ev.archivo) rec.archivo = ev.archivo;
  if (ev.hoja) rec.hoja = ev.hoja;
  if (backupKey) rec.backup_lote_key = backupKey;
  if (createdAt && (!rec.created_at || createdAt < rec.created_at)) rec.created_at = createdAt;
  if (createdAt && (!rec.lote_created_at || createdAt < rec.lote_created_at)) rec.lote_created_at = createdAt;
  const lastTs = eventTimestamp_(ev);
  if (lastTs && (!rec.ultimo_evento || lastTs > rec.ultimo_evento)) rec.ultimo_evento = lastTs;
  if (String(ev.total_lineas || ev.productos_total || "").trim()) rec.total_lineas = ev.total_lineas || ev.productos_total;
  if (String(ev.total_unidades || ev.unidades_total || "").trim()) rec.total_unidades = ev.total_unidades || ev.unidades_total;
  if (ev.status || ev.estado) rec.status = ev.status || ev.estado;
  catalogMap[rescueKey] = rec;
}

function appendMissingLotesFromMain_(ss, catalogMap) {
  const sh = ss.getSheetByName("eventos");
  if (!sh || sh.getLastRow() < 2) return;
  const headers = getHeaders_(sh);
  const typeIndex = headers.indexOf("event_type");
  if (typeIndex < 0) return;
  const finder = sh.getRange(2, typeIndex + 1, sh.getLastRow() - 1, 1)
    .createTextFinder("lote_creado")
    .matchEntireCell(true);
  const cells = finder.findAll() || [];
  cells.forEach(function(cell) {
    const rowValues = sh.getRange(cell.getRow(), 1, 1, headers.length).getValues()[0];
    const row = {};
    headers.forEach(function(h, i) { row[h] = rowValues[i]; });
    addCatalogEvent_(catalogMap, parseRawMerge_(row));
  });
}

function buildLoteCatalog_(ss) {
  const catalogMap = {};
  const loteSheet = ss.getSheetByName("lotes");
  if (loteSheet) {
    sheetToObjects_(loteSheet).forEach(function(row) {
      addCatalogEvent_(catalogMap, parseRawMerge_(row));
    });
  }
  // Si writeStructured_ falló en algún momento, lote_creado sigue existiendo en eventos.
  appendMissingLotesFromMain_(ss, catalogMap);

  const records = Object.keys(catalogMap).map(function(k) { return catalogMap[k]; });
  const stateSheet = ss.getSheetByName("lotes_estados");
  if (stateSheet) {
    sheetToObjects_(stateSheet).forEach(function(row) {
      const ev = parseRawMerge_(row);
      const rec = catalogMatch_(records, ev);
      if (!rec) return;
      const et = normalizedText_(ev.event_type || "");
      if (et === "LOTE_CERRADO") rec.status = "CERRADO";
      else if (et === "LOTE_REABIERTO") rec.status = "ACTIVO";
      else if (et === "LOTE_ELIMINADO") rec.status = "ELIMINADO";
      else if (ev.status) rec.status = ev.status;
      const ts = eventTimestamp_(ev);
      if (ts && (!rec.ultimo_evento || ts > rec.ultimo_evento)) rec.ultimo_evento = ts;
    });
  }

  // Delimitar versiones que reutilizaron el mismo lote_id.
  const byId = {};
  records.forEach(function(rec) {
    const key = String(rec.lote_id || "");
    if (!byId[key]) byId[key] = [];
    byId[key].push(rec);
  });
  Object.keys(byId).forEach(function(key) {
    const arr = byId[key];
    arr.sort(function(a, b) { return timestampText_(a.created_at).localeCompare(timestampText_(b.created_at)); });
    arr.forEach(function(rec, index) {
      rec.scope_start = timestampText_(rec.created_at);
      rec.scope_end = index + 1 < arr.length ? timestampText_(arr[index + 1].created_at) : "";
    });
  });

  records.sort(function(a, b) {
    return timestampText_(b.ultimo_evento || b.created_at).localeCompare(timestampText_(a.ultimo_evento || a.created_at));
  });
  return records;
}

function eventMatchesLote_(ev, params) {
  const requestedLoteId = String(params.lote_id || "").trim();
  if (requestedLoteId && String(ev.lote_id || "").trim() !== requestedLoteId) return false;
  if (!requestedLoteId && !params.backup_lote_key && !params.lote_nombre && !params.archivo && !params.hoja) return true;

  const requestedKey = String(params.backup_lote_key || "").trim();
  const eventKey = String(ev.backup_lote_key || "").trim();
  if (requestedKey && eventKey) return requestedKey === eventKey;

  const checks = [
    [params.lote_nombre, ev.lote_nombre],
    [params.archivo, ev.archivo],
    [params.hoja, ev.hoja]
  ];
  let positiveIdentity = false;
  for (let i = 0; i < checks.length; i++) {
    const wanted = normalizedText_(checks[i][0] || "");
    const actual = normalizedText_(checks[i][1] || "");
    if (wanted && actual) {
      if (wanted !== actual) return false;
      positiveIdentity = true;
    }
  }

  const ts = eventTimestamp_(ev);
  const start = timestampText_(params.scope_start || params.lote_created_at || "");
  const end = timestampText_(params.scope_end || "");
  if (start && ts && ts < start) return false;
  if (end && ts && ts >= end) return false;
  if (start && ts) positiveIdentity = true;

  // Con backup key solicitado, los eventos antiguos sin esa columna solo entran si
  // coinciden por metadatos o por la ventana temporal exacta del FULL.
  return positiveIdentity || (!requestedKey && !!requestedLoteId);
}

function readMainEventsPage_(ss, params) {
  const sh = getOrCreateSheet_(ss, "eventos", EVENT_HEADERS);
  ensureHeaders_(sh, EVENT_HEADERS);
  const headers = getHeaders_(sh);
  const lastRow = sh.getLastRow();
  const pageSize = clampInt_(params.page_size || 750, 100, 1500);
  const scanRows = clampInt_(params.scan_rows || 6000, 1000, 10000);
  let cursor = clampInt_(params.cursor || 2, 2, Math.max(2, lastRow + 1));
  const events = [];

  if (lastRow < 2 || cursor > lastRow) {
    return { ok: true, events: events, next_cursor: "", done: true, scanned_until: lastRow };
  }

  const endRow = Math.min(lastRow, cursor + scanRows - 1);
  const values = sh.getRange(cursor, 1, endRow - cursor + 1, headers.length).getValues();
  let nextRow = cursor;
  for (let i = 0; i < values.length; i++) {
    const rowNumber = cursor + i;
    const row = {};
    headers.forEach(function(h, col) { row[h] = values[i][col]; });
    const ev = parseRawMerge_(row);
    nextRow = rowNumber + 1;
    if (eventMatchesLote_(ev, params)) events.push(ev);
    if (events.length >= pageSize) break;
  }

  const hasMore = nextRow <= lastRow;
  return {
    ok: true,
    events: events,
    next_cursor: hasMore ? String(nextRow) : "",
    done: !hasMore,
    scanned_until: nextRow - 1,
    last_row: lastRow
  };
}

function readRecentEventKeys_(ss, limit) {
  const sh = getOrCreateSheet_(ss, "event_keys", EVENT_KEY_HEADERS);
  if (sh.getLastRow() < 2) return [];
  const headers = getHeaders_(sh);
  const count = Math.min(clampInt_(limit || 5000, 100, 20000), sh.getLastRow() - 1);
  const start = Math.max(2, sh.getLastRow() - count + 1);
  const values = sh.getRange(start, 1, count, headers.length).getValues();
  return values.map(function(row) {
    const obj = {};
    headers.forEach(function(h, i) { obj[h] = row[i]; });
    return { event_key: obj.event_key || "", event_uid: obj.event_uid || "" };
  });
}

function appendStructuredPicking_(ss, events) {
  const lsh = ss.getSheetByName("picking_listas");
  const ish = ss.getSheetByName("picking_items");
  if (!lsh && !ish) return;

  const listRows = lsh ? sheetToObjects_(lsh) : [];
  const itemRows = ish ? sheetToObjects_(ish) : [];
  const itemsByCode = {};
  itemRows.forEach(function(r) {
    const key = pickingKey_(r);
    if (!key) return;
    if (!itemsByCode[key]) itemsByCode[key] = [];
    itemsByCode[key].push({
      item_id: r.item_id || "",
      codigo_ml: r.codigo_ml || "",
      codigo_universal: r.codigo_universal || "",
      sku: r.sku || "",
      descripcion: r.descripcion || "",
      descripcion_kame: r.descripcion_kame || r.descripcion || "",
      descripcion_ml: r.descripcion_ml || "",
      familia_kame: r.familia_kame || "",
      maestro_match_status: r.maestro_match_status || "",
      cantidad: r.cantidad || "",
      area: r.area || "",
      nro: r.nro || ""
    });
  });

  const seenListKeys = {};
  listRows.forEach(function(r) {
    const key = pickingKey_(r);
    if (!key) return;
    seenListKeys[key] = true;
    const et = r.event_type || eventTypeFromPickingStatus_(r.estado || "CREADA");
    events.push({
      event_type: et,
      queue_id: r.queue_id || "STRUCT_PICKING_LISTA",
      queued_at: r.created_at || r.received_at || "",
      created_at: r.created_at || r.received_at || "",
      lote_id: r.lote_id || "",
      lote_nombre: r.lote_nombre || "",
      picking_list_id: r.picking_list_id || "",
      picking_code: r.picking_code || r.codigo_lista || "",
      codigo_lista: r.codigo_lista || r.picking_code || "",
      asignado_a: r.asignado_a || "",
      estado: r.estado || "",
      created_by: r.created_by || "",
      usuario: r.usuario || "",
      comentario: r.comentario || "",
      productos: r.productos || "",
      cantidad: r.cantidad || "",
      items: itemsByCode[key] || [],
      event_key: r.event_key || buildEventKey_(r)
    });
  });

  // Si hay picking_items pero faltó picking_listas, crear lista sintética recuperable.
  Object.keys(itemsByCode).forEach(function(key) {
    if (seenListKeys[key]) return;
    const first = itemRows.find(function(r) { return pickingKey_(r) === key; }) || {};
    events.push({
      event_type: "picking_lista_creada",
      queue_id: first.queue_id || "STRUCT_PICKING_ITEMS_ONLY",
      queued_at: first.created_at || first.received_at || "",
      created_at: first.created_at || first.received_at || "",
      lote_id: first.lote_id || "",
      lote_nombre: first.lote_nombre || "",
      picking_list_id: first.picking_list_id || "",
      picking_code: first.picking_code || first.codigo_lista || "",
      codigo_lista: first.codigo_lista || first.picking_code || "",
      asignado_a: first.asignado_a || "",
      estado: first.estado_lista || "CREADA",
      items: itemsByCode[key],
      event_key: "STRUCT_PICKING_ITEMS_ONLY:" + key
    });
  });
}

function eventTypeFromPickingStatus_(estado) {
  const e = String(estado || "").toUpperCase();
  if (e === "IMPRESA") return "picking_lista_impresa";
  if (e === "COMPLETADA") return "picking_lista_completada";
  if (e === "ANULADA") return "picking_lista_anulada";
  return "picking_lista_creada";
}

function pickingKey_(r) {
  const lote = String(r.lote_id || "").trim();
  const code = String(r.picking_code || r.codigo_lista || r.picking_list_id || "").trim();
  if (!lote || !code) return "";
  return lote + "|" + code;
}

function buildEventKey_(data) {
  data = data || {};
  const eventUid = String(data.event_uid || "").trim();
  if (eventUid) return "UID:" + eventUid;

  return [
    data.event_type || data.source_event_type || "",
    data.queue_id || "",
    data.queued_at || "",
    data.created_at || "",
    data.lote_id || "",
    data.picking_code || data.codigo_lista || "",
    data.picking_list_id || "",
    data.print_scope || data.scope || "",
    data.print_kind || "",
    data.block_index || "",
    data.block_key || "",
    data.zpl_hash || "",
    data.item_id || "",
    data.codigo_ml || "",
    data.codigo_universal || "",
    data.sku || "",
    data.cantidad || "",
    data.tipo || data.tipo_aviso || "",
    data.comentario || data.detail || ""
  ].map(function(x) { return String(x === null || x === undefined ? "" : x).trim(); }).join("|");
}

function eventAlreadyExists_(ss, eventos, eventKey) {
  if (!eventKey) return false;
  const keySheet = getOrCreateSheet_(ss, "event_keys", EVENT_KEY_HEADERS);

  // No reconstruimos el índice completo dentro de doPost.
  // La primera versión podía demorar demasiado y causar timeout en Streamlit,
  // aunque Sheets sí alcanzara a escribir el evento.
  // Para la operación diaria basta revisar el índice y las últimas filas de eventos.
  return findEventKeyInSheet_(keySheet, eventKey) || findEventKeyInRecentEvents_(eventos, eventKey, 500);
}

function rememberEventKey_(ss, data, eventKey) {
  if (!eventKey) return;
  const keySheet = getOrCreateSheet_(ss, "event_keys", EVENT_KEY_HEADERS);
  if (findEventKeyInSheet_(keySheet, eventKey)) return;
  appendObjectRow_(keySheet, {
    event_key: eventKey,
    event_uid: data.event_uid || "",
    event_source: data.event_source || "",
    first_seen_at: new Date(),
    event_type: data.event_type || data.source_event_type || "",
    lote_id: data.lote_id || "",
    item_id: data.item_id || "",
    sku: data.sku || "",
    created_at: data.created_at || data.queued_at || ""
  });
}

function ensureEventKeyIndex_(eventos, keySheet, force) {
  if (!eventos || !keySheet) return 0;
  if (!force && keySheet.getLastRow() >= 2) return 0;
  if (eventos.getLastRow() < 2) return 0;

  const eventRows = sheetToObjects_(eventos);
  const headers = getHeaders_(keySheet);
  const rows = [];
  const seen = {};

  eventRows.forEach(function(r) {
    const merged = parseRawMerge_(r);
    const key = String(merged.event_key || buildEventKey_(merged) || "").trim();
    if (!key || seen[key]) return;
    seen[key] = true;
    const obj = {
      event_key: key,
      event_uid: merged.event_uid || "",
      event_source: merged.event_source || "",
      first_seen_at: merged.received_at || new Date(),
      event_type: merged.event_type || merged.source_event_type || "",
      lote_id: merged.lote_id || "",
      item_id: merged.item_id || "",
      sku: merged.sku || "",
      created_at: merged.created_at || merged.queued_at || ""
    };
    rows.push(headers.map(function(h) {
      const v = obj[h];
      return v === undefined || v === null ? "" : v;
    }));
  });

  if (!rows.length) return 0;

  const startRow = keySheet.getLastRow() + 1;
  keySheet.getRange(startRow, 1, rows.length, headers.length).setValues(rows);
  return rows.length;
}

function findEventKeyInRecentEvents_(sheet, eventKey, maxRows) {
  if (!sheet || !eventKey) return false;
  const headers = getHeaders_(sheet);
  const idx = headers.indexOf("event_key");
  const lastRow = sheet.getLastRow();
  if (idx < 0 || lastRow < 2) return false;

  const n = Math.min(Number(maxRows || 500), lastRow - 1);
  const start = Math.max(2, lastRow - n + 1);
  const values = sheet.getRange(start, idx + 1, n, 1).getValues().flat();
  return values.map(String).indexOf(String(eventKey)) !== -1;
}

function findEventKeyInSheet_(sheet, eventKey) {
  if (!sheet || !eventKey) return false;
  const headers = getHeaders_(sheet);
  const idx = headers.indexOf("event_key");
  const lastRow = sheet.getLastRow();
  if (idx < 0 || lastRow < 2) return false;
  const range = sheet.getRange(2, idx + 1, lastRow - 1, 1);
  return range.createTextFinder(String(eventKey)).matchEntireCell(true).findNext() !== null;
}

function getOrCreateSheet_(ss, name, headers) {
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  if (sh.getLastRow() === 0) sh.appendRow(headers);
  ensureHeaders_(sh, headers);
  return sh;
}

function ensureHeaders_(sheet, headers) {
  const current = getHeaders_(sheet);
  if (current.length === 0) {
    sheet.appendRow(headers);
    return;
  }
  headers.forEach(function(h) {
    if (current.indexOf(h) === -1) {
      sheet.getRange(1, sheet.getLastColumn() + 1).setValue(h);
      current.push(h);
    }
  });
}

function getHeaders_(sheet) {
  if (!sheet || sheet.getLastRow() < 1 || sheet.getLastColumn() < 1) return [];
  return sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(function(h) { return String(h || "").trim(); });
}

function appendObjectRow_(sheet, obj) {
  const headers = getHeaders_(sheet);
  const row = headers.map(function(h) {
    const v = obj[h];
    return v === undefined || v === null ? "" : v;
  });
  sheet.appendRow(row);
}

function sheetToObjects_(sheet) {
  if (!sheet || sheet.getLastRow() < 2) return [];
  const values = sheet.getDataRange().getValues();
  const headers = values[0].map(function(h) { return String(h || "").trim(); });
  const out = [];
  for (let r = 1; r < values.length; r++) {
    const obj = {};
    headers.forEach(function(h, i) { obj[h] = values[r][i]; });
    out.push(obj);
  }
  return out;
}

function parseRawMerge_(row) {
  const out = Object.assign({}, row || {});
  if (out.raw_json) {
    try {
      const parsed = JSON.parse(out.raw_json);
      if (parsed && typeof parsed === "object") {
        Object.keys(parsed).forEach(function(k) { out[k] = parsed[k]; });
      }
    } catch (err) {}
  }
  return out;
}

function logError_(source, err, e) {
  try {
    const ss = SpreadsheetApp.openById(ERRORS_SPREADSHEET_ID);
    const sh = getOrCreateSheet_(ss, "errores", ["created_at", "source", "error", "payload"]);
    appendObjectRow_(sh, {
      created_at: new Date(),
      source: source,
      error: String(err),
      payload: e && e.postData ? e.postData.contents : JSON.stringify(e || {})
    });
  } catch (innerErr) {}
}

function json_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
